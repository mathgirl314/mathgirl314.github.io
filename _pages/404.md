You will be redirected to the main page within <strong id="countdown">3</strong> seconds. If not redirected, please go back to the [home page]({{ site.baseurl | prepend: site.url }}).

<script>
  let seconds = 3;
  const countdownEl = document.getElementById('countdown');
  const targetUrl = "{{ site.baseurl | prepend: site.url }}/";

  const interval = setInterval(() => {
    seconds--;
    if (countdownEl) {
      countdownEl.textContent = seconds;
    }
    if (seconds <= 0) {
      clearInterval(interval);
      window.location.href = targetUrl;
    }
  }, 1000);
</script>
